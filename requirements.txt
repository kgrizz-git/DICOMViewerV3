# DICOM Viewer V3 Requirements
# 
# This file lists all Python package dependencies required for the DICOM viewer application.
# 
# Installation: pip install -r requirements.txt
#
# Python version (especially Windows):
# - The full stack requires Python >= 3.12 because numpy>=2.5.1 requires it.
# - Python 3.12 is the CI and recommended version. Wheels exist for native deps (e.g. pyjpegls).
# - 3.14+ may have no pyjpegls wheel on Windows yet; pip then builds from source and needs
#   Microsoft C++ Build Tools (https://visualstudio.microsoft.com/visual-cpp-build-tools/).
# - Shared-folder paths can confuse venv/pip; a local clone is often more reliable if you see
#   odd path or pip self-upgrade issues.
#
# Requirements:
# - PySide6: Cross-platform GUI framework for the application interface
# - pydicom: Library for reading, writing, and working with DICOM files
# - numpy: Numerical computing library for image array operations
# - Pillow: Image processing library for image format conversions
# - matplotlib: Plotting library for histogram display
# - openpyxl: Library for reading and writing Excel files (for tag export)

PySide6>=6.11.1
# Stays on pydicom 2.x: pylinac 3.43.2 caps it at `pydicom<3`. This also blocks the
# imagecodecs (BSD) decoder plugin and highdicom, which need pydicom 3.x. Moving to
# pydicom 3 requires a coordinated pylinac bump + ACR QA re-verify. See
# dev-docs/info/HIGHDICOM_OVERVIEW.md and PYLIBJPEG_ALTERNATIVES_AND_DICOM_DECODER_STRATEGY.md.
pydicom>=2.4.5,<3
numpy>=2.5.1
Pillow>=12.3.0
# Cine export (GIF / AVI / MPG): imageio + bundled FFmpeg via imageio-ffmpeg (LGPL/GPL components — see CHANGELOG [Unreleased] and README).
imageio==2.37.4
imageio-ffmpeg==0.6.0
matplotlib>=3.11.1
openpyxl>=3.1.5
# Local encrypted study index (SQLCipher-compatible Python binding; wheels on Windows/macOS/Linux for supported Python versions)
sqlcipher3>=0.6.2
# Stores the SQLCipher passphrase in the OS credential store (Windows Credential Manager, macOS Keychain, etc.)
keyring>=25.7.0
SimpleITK>=2.5.5  # For 3D image resampling in image fusion (Phase 2)
vtk>=9.6.2  # GPU-accelerated 3D volume rendering (vtkSmartVolumeMapper + QVTKRenderWindowInteractor)
# Pinned exact version; 3.43.2 is the current verified ACR CT/MRI integration pin. Re-verify after each bump (see dev-docs/info/PYLINAC_INTEGRATION_OVERVIEW.md, dev-docs/plans/DEPENDENCY_BUMP_VERIFICATION_PLAN.md).
pylinac==3.43.2  # Automated phantom QA analysis (ACR CT + ACR MRI Large)
scipy>=1.18.1  # pylinac numerical dependency
scikit-image>=0.22.0  # pylinac image processing dependency
pypdf>=6.16.2  # PDF merging for compare-mode combined PDF report
# HTTP(S) client for local SonarQube utilities and public update metadata checks.
requests>=2.32.5

# Compressed DICOM support
# `python-gdcm` is the selected classic JPEG Baseline/Extended/Lossless decoder.
# Keep the independently verified release pin until the productionization plan's
# cross-platform dependency-bump verification is complete.
pylibjpeg>=2.1.0  # For JPEG 2000 and other compressed formats
python-gdcm==3.2.6  # Classic JPEG Baseline/Extended/Lossless pydicom handler
pylibjpeg-openjpeg>=2.5.0  # JPEG 2000 decoder plugin for pylibjpeg
pylibjpeg-rle>=2.2.0  # RLE decoder plugin for pylibjpeg
pyjpegls>=1.5.1  # For JPEG-LS compression

# XXE-hardened XML parsing. Used by scripts/new_code_coverage.py (reads
# coverage.xml); in requirements.txt so the src+scripts type-check and every CI
# job resolve it. Tiny, pure-Python, permissively licensed.
defusedxml>=0.7.1

# Testing (recommended for project venv; run: pip install -r requirements.txt)
pytest>=9.1.1  # Test runner; supports tests/ and unittest-style tests
# Floor raised through 2.21.0; still covers AdlLexer issue in pygments<2.20.0
# (transitive via pytest, rich).
Pygments>=2.21.0
